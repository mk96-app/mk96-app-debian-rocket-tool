# Recordatorios de reuniones / calendario local — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a local, manual calendar of meetings/reminders to the tray app: create meetings with date, time, simple recurrence, lead-time alert, link, and notes; get sound + desktop notification + a persistent tray reminder when one is due.

**Architecture:** A new pure-logic module (`meetings.py`) models meetings and computes recurrence/due-alerts, mirroring the existing `rules.py`/`poller.py` split between pure functions and GLib-driven orchestration. A new `MeetingScheduler` (mirrors `Poller`) ticks independently of the Rocket.Chat connection. A new `meetings_window.py` (mirrors `settings_window.py`) manages CRUD. `tray.py` gains pending/upcoming meeting sections. Shared GTK dialog code (`show_error_dialog`, `open_sound_picker`) is extracted from `settings_window.py` into `gtk_helpers.py` so both windows can use it without duplication.

**Tech Stack:** Python 3.10, PyGObject (GTK3, GLib, AyatanaAppIndicator3), PyYAML, pytest. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-08-18-meetings-design.md`

A meeting dict (post-`normalize_meeting`) always has exactly these keys: `id`, `titulo`, `fecha_hora` (str `"YYYY-MM-DDTHH:MM"`), `repeticion_modo` (`"ninguna"|"diaria"|"semanal"|"mensual"`), `repeticion_dias` (list of codes from `DIAS_SEMANA`, only meaningful for `"semanal"`), `repeticion_fin` (str `"YYYY-MM-DD"` or `None`), `aviso_antes_min` (int), `enlace` (str or `None`), `notas` (str), `sonido_id` (str or `None`), `activa` (bool).

**Important gotcha to handle:** PyYAML's `safe_load` auto-converts bare date-looking scalars (e.g. `2026-12-31`) into `datetime.date` objects, but leaves `"2026-08-20T10:00"` (no seconds) as a plain string, because its timestamp regex requires seconds. This means `repeticion_fin` may arrive as either a `str` or a `datetime.date` depending on whether the YAML author quoted it. `normalize_meeting` must coerce it to a plain `"YYYY-MM-DD"` string either way, so the rest of the code never has to care.

```python
"""Calendario local de reuniones: modelo de datos y lógica pura de recurrencia/avisos.

Funciones puras (sin GLib/GTK/I-O), testeables directamente -mismo patrón que
rules.py/poller.py. La orquestación con efectos (temporizador, sonido, notificación)
vive en meeting_scheduler.py.
"""

import uuid
from datetime import date, datetime, timedelta

DIAS_SEMANA = ["lun", "mar", "mie", "jue", "vie", "sab", "dom"]
FECHA_HORA_FORMAT = "%Y-%m-%dT%H:%M"
FECHA_FORMAT = "%Y-%m-%d"

# Sonido de la paleta empaquetada (ver sounds/default/LICENSES.md) usado cuando una
# reunión no tiene sonido_id propio.
DEFAULT_MEETING_SONIDO_ID = "interface-sounds/confirmation_001"

# Tope defensivo para el modo "mensual": evita un bucle infinito si algún día
# repeticion_fin/fecha_hora quedaran mal formados.
MAX_MESES_BUSQUEDA = 60


def _fecha_str(value) -> str | None:
    """PyYAML auto-convierte escalares con pinta de fecha ('2026-12-31') a
    datetime.date al cargar el YAML -esto normaliza cualquiera de las dos formas
    a un str 'YYYY-MM-DD', para que el resto del código nunca tenga que
    distinguir entre las dos representaciones."""
    if value is None:
        return None
    if isinstance(value, (date, datetime)):
        return value.strftime(FECHA_FORMAT)
    return value


def normalize_meeting(raw: dict) -> dict:
    """Rellena los campos opcionales de una reunión con sus valores por defecto,
    igual que normalize_rule en config.py."""
    return {
        "id": raw.get("id") or str(uuid.uuid4()),
        "titulo": raw["titulo"],
        "fecha_hora": raw["fecha_hora"],
        "repeticion_modo": raw.get("repeticion_modo", "ninguna"),
        "repeticion_dias": raw.get("repeticion_dias") or [],
        "repeticion_fin": _fecha_str(raw.get("repeticion_fin")),
        "aviso_antes_min": raw.get("aviso_antes_min", 0),
        "enlace": raw.get("enlace"),
        "notas": raw.get("notas", ""),
        "sonido_id": raw.get("sonido_id"),
        "activa": raw.get("activa", True),
    }


def _parse_fecha_hora(fecha_hora: str) -> datetime:
    return datetime.strptime(fecha_hora, FECHA_HORA_FORMAT)


def _parse_fecha_fin(repeticion_fin: str | None):
    if repeticion_fin is None:
        return None
    return datetime.strptime(repeticion_fin, FECHA_FORMAT).date()


def _pasada_fecha_fin(candidate_date, repeticion_fin_date) -> bool:
    return repeticion_fin_date is not None and candidate_date > repeticion_fin_date


def next_occurrence(meeting: dict, ahora: datetime) -> datetime | None:
    """Próxima fecha/hora de la reunión en el instante `ahora` o posterior (>=),
    respetando repeticion_modo/repeticion_dias/repeticion_fin. `None` si la reunión
    está inactiva o la serie ya ha terminado."""
    if not meeting.get("activa", True):
        return None

    base = _parse_fecha_hora(meeting["fecha_hora"])
    repeticion_fin_date = _parse_fecha_fin(meeting.get("repeticion_fin"))
    modo = meeting.get("repeticion_modo", "ninguna")

    if modo == "ninguna":
        return base if base >= ahora else None

    if modo == "diaria":
        start_date = max(base.date(), ahora.date())
        candidate = datetime.combine(start_date, base.time())
        if candidate < ahora:
            candidate += timedelta(days=1)
        if _pasada_fecha_fin(candidate.date(), repeticion_fin_date):
            return None
        return candidate

    if modo == "semanal":
        dias = meeting.get("repeticion_dias") or []
        start_date = max(base.date(), ahora.date())
        for offset in range(8):
            candidate_date = start_date + timedelta(days=offset)
            if DIAS_SEMANA[candidate_date.weekday()] not in dias:
                continue
            candidate = datetime.combine(candidate_date, base.time())
            if candidate < ahora:
                continue
            if _pasada_fecha_fin(candidate_date, repeticion_fin_date):
                return None
            return candidate
        return None

    if modo == "mensual":
        year, month = base.year, base.month
        for _ in range(MAX_MESES_BUSQUEDA):
            try:
                candidate_date = date(year, month, base.day)
            except ValueError:
                candidate_date = None  # ese mes no tiene ese día -se omite

            if candidate_date is not None:
                candidate = datetime.combine(candidate_date, base.time())
                if candidate >= base and candidate >= ahora:
                    if _pasada_fecha_fin(candidate_date, repeticion_fin_date):
                        return None
                    return candidate

            month += 1
            if month > 12:
                month = 1
                year += 1
        return None

    raise ValueError(f"repeticion_modo desconocido: {modo!r}")


def due_meetings(meetings: list, ahora: datetime, ultimo_chequeo: datetime) -> list:
    """Reuniones cuyo aviso (ocurrencia - aviso_antes_min) cae en la ventana
    semiabierta (ultimo_chequeo, ahora] -evita duplicar el aviso entre ticks
    consecutivos del planificador. Devuelve [(meeting, ocurrencia), ...]."""
    resultado = []
    for meeting in meetings:
        ocurrencia = next_occurrence(meeting, ultimo_chequeo)
        if ocurrencia is None:
            continue
        aviso_en = ocurrencia - timedelta(minutes=meeting.get("aviso_antes_min", 0))
        if ultimo_chequeo < aviso_en <= ahora:
            resultado.append((meeting, ocurrencia))
    return resultado
```

Rocket.Chat -funciona sin servidor conectado. La lógica de qué reuniones están debidas
vive en meetings.due_meetings (pura); esta clase solo la orquesta con el temporizador.

from datetime import datetime

import gi

gi.require_version("GLib", "2.0")
from gi.repository import GLib

from mk96_app_debian_rocket_tool.meetings import due_meetings

DEFAULT_CHECK_INTERVAL_S = 30


class MeetingScheduler:
    def __init__(self, get_meetings, on_due, on_tick=None, check_interval_s=DEFAULT_CHECK_INTERVAL_S):
        """get_meetings() -> list de reuniones actuales (se llama en cada tick, para ver
        siempre la config más reciente). on_due(meeting, ocurrencia) se llama por cada
        reunión cuyo aviso toca disparar. on_tick(), si se da, se llama en cada tick tras
        procesar los avisos -para refrescar UI derivada (p.ej. "próximas reuniones")
        aunque no haya ningún aviso nuevo."""
        self._get_meetings = get_meetings
        self._on_due = on_due
        self._on_tick = on_tick or (lambda: None)
        self._check_interval_s = check_interval_s
        self._ultimo_chequeo = None
        self._timeout_id = None

    def start(self) -> None:
        self._ultimo_chequeo = datetime.now()
        self._timeout_id = GLib.timeout_add_seconds(self._check_interval_s, self._tick)

    def stop(self) -> None:
        if self._timeout_id is not None:
            GLib.source_remove(self._timeout_id)
            self._timeout_id = None

    def _tick(self) -> bool:
        ahora = datetime.now()
        for meeting, ocurrencia in due_meetings(self._get_meetings(), ahora, self._ultimo_chequeo):
            self._on_due(meeting, ocurrencia)
        self._ultimo_chequeo = ahora
        self._on_tick()
        return GLib.SOURCE_CONTINUE

        if self._selected_sonido_id is None:
            show_error_dialog(

            )
            return
        self._preview_sonido_id(self._selected_sonido_id)

    def _preview_sonido_id(self, sonido_id: str) -> None:
        try:
            sound_manager.play_sound(sonido_id)
        except sound_manager.SoundPlaybackError as exc:
            show_error_dialog(self, "No se pudo reproducir el sonido", str(exc))

    def _choose_sound(self) -> None:
        chosen = open_sound_picker(
            self, self._selected_sonido_id, self._preview_sonido_id,
            lambda summary, body: show_error_dialog(self, summary, body),
        )
        if chosen is not None:
            self._selected_sonido_id = chosen
            self._sonido_label.set_text(sonido_display_name(chosen))

    # -- Guardar ---------------------------------------------------------------------

    def _save_meeting(self) -> None:
        titulo = self._titulo_entry.get_text().strip()
        fecha = self._fecha_entry.get_text().strip()
        hora = self._hora_entry.get_text().strip()
        if not titulo or not fecha or not hora:
            show_error_dialog(self, "Faltan datos", "Título, fecha y hora son obligatorios.")
            return

        try:
            datetime.strptime(f"{fecha}T{hora}", "%Y-%m-%dT%H:%M")
        except ValueError:
            show_error_dialog(
                self, "Fecha u hora inválida",
                "Usa el formato AAAA-MM-DD para la fecha y HH:MM para la hora.",
            )
            return

        repeticion_fin = self._fin_entry.get_text().strip() or None
        if repeticion_fin is not None:
            try:
                datetime.strptime(repeticion_fin, "%Y-%m-%d")
            except ValueError:
                show_error_dialog(self, "Fecha de fin inválida", "Usa el formato AAAA-MM-DD.")
                return

        modo = self._repeticion_combo.get_active_id()
        dias = [dia for dia, check in self._dia_checks.items() if check.get_active()]
        if modo == "semanal" and not dias:
            show_error_dialog(
                self, "Faltan días", "Elige al menos un día de la semana para la repetición semanal."
            )
            return

        meeting = normalize_meeting(
            {
                "id": self._editing_id,
                "titulo": titulo,
                "fecha_hora": f"{fecha}T{hora}",
                "repeticion_modo": modo,
                "repeticion_dias": dias,
                "repeticion_fin": repeticion_fin,
                "aviso_antes_min": int(self._aviso_spin.get_value()),
                "enlace": self._enlace_entry.get_text().strip() or None,
                "notas": self._notas_entry.get_text().strip(),
                "sonido_id": self._selected_sonido_id,
                "activa": self._activa_check.get_active(),
            }
        )

        self._config["reuniones"] = [
            m for m in self._config["reuniones"] if m["id"] != meeting["id"]
        ] + [meeting]
        self._on_config_changed(self._config)
        self._refresh_list()
        self._clear_form()
